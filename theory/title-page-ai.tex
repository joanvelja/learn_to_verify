\newcommand{\addtodotostorage}[3]{%
  % #1 = user, #2 = text, #3 = type (margin/inline)
  \stepcounter{todocount}%

}

% ─── TODO COMMANDS ─────────────────────────────────────────────────
% Margin-note
\NewDocumentCommand{\todo}{O{Anonymous} m}{%
  \addtodotostorage{#1}{#2}{margin}%
  \begingroup
    \def\user{#1}%
    \marginpar{%
      \begin{tikzpicture}[baseline=(current bounding box.north)]
        \node[
          draw=\getUserColor{\user},
          fill={\getUserColor{\user}!10},
          rounded corners=3pt,
          text width=\marginparwidth-8pt,
          inner sep=4pt,
          font=\footnotesize
        ] (box) {%
          \textbf{\textcolor{\getUserColor{\user}}{\user:}}~#2%
        };
        \fill[\getUserColor{\user}]
          ([xshift=-2pt]box.west) --
          ([xshift=-6pt,yshift= 2pt]box.west) --
          ([xshift=-6pt,yshift=-2pt]box.west) -- cycle;
      \end{tikzpicture}%
    }%
  \endgroup
}
% Inline
\NewDocumentCommand{\todoinline}{O{Anonymous} m}{%
  \addtodotostorage{#1}{#2}{inline}%
  \begingroup
    \def\user{#1}%
    \tikz[baseline=(txt.base)]{
      \node[
        draw=\getUserColor{\user},
        fill={\getUserColor{\user}!15},
        rounded corners=2pt,
        inner sep=2pt,
        font=\scriptsize
      ] (txt) {\textbf{\textcolor{\getUserColor{\user}}{\user:}}~#2};
    }%
  \endgroup
}

% ─── LIST TODOS COMMAND ─────────────────────────────────────────────
\newcommand{\listtodousers}{%
  \section*{TODO Summary}%
  \ifx\todolist\empty
    \textit{No todos found.}
  \else
    % First pass: collect unique users
    \def\uniqueusers{}%
    \forlistloop{\collectusers}{\todolist}%

    % Second pass: display todos by user
    \renewcommand*{\do}[1]{%
      \subsection*{\textcolor{\getUserColor{##1}}{\bfseries ##1}}%
      \begin{itemize}%
        \forlistloop{\displaytodos{##1}}{\todolist}%
      \end{itemize}%
      \vspace{0.5em}%
    }%
    \dolistloop{\uniqueusers}%

    \vspace{0.5em}%
    \noindent{\small\textit{Legend:} \textcolor{gray}{\textbullet} = Margin note, \textcolor{black}{\circ} = \text{Inline note}}%
  \fi
}

% Helper commands for list processing
\newcommand{\collectusers}[1]{%
  \expandafter\collectusersaux#1%
}
\newcommand{\collectusersaux}[3]{%
  % #1 = user, #2 = text, #3 = type
  \xifinlist{#1}{\uniqueusers}{}{%
    \listxadd{\uniqueusers}{#1}%
  }%
}

\newcommand{\displaytodos}[2]{%
  % #1 = target user, #2 = todo entry
  \expandafter\displaytodosaux\expandafter{#1}#2%
}
\newcommand{\displaytodosaux}[4]{%
  % #1 = target user, #2 = todo user, #3 = text, #4 = type
  \ifstrequal{#1}{#2}{%
    \item[\textcolor{\getUserColor{#2}}{\ifstrequal{#4}{margin}{\textbullet}{\circ}}] #3%
  }{}%
}

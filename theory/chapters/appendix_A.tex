\subsection{Formal setup and notation}

\paragraph{Measurable spaces.}
Let $(\mathcal{X}, \mathcal{B}_X)$ and $(\mathcal{Z}, \mathcal{B}_Z)$ be Polish spaces (e.g., for text, the space of all finite-length strings) equipped with their Borel $\sigma$-algebras. Let $P_X$ be a probability measure over the problem space $\mathcal{X}$. We assume $\mathcal X,\mathcal Z$ are Polish such that standard measurable-selection results apply; in the discrete code-benchmark actually used, this assumption is automatically satisfied.

\paragraph{Policies and Verifier.}
We consider three deterministic, measurable mappings:
\smallskip
\begin{itemize}
\item A \textbf{helpful prover} $\pi_H: \mathcal{X} \to \mathcal{Z}$, which aims to produce correct solutions.
\item A \textbf{sneaky prover} $\pi_S: \mathcal{X} \times \mathcal{Z} \to \mathcal{Z}$, which observes the helpful prover's output and aims to produce an incorrect solution that appears convincing.
\item A \textbf{verifier} $V: \mathcal{X} \times \mathcal{Z} \to [-B, B]$, which outputs a score indicating its belief that a solution comes from the honest prover.
\end{itemize}

\paragraph{Policy classes and measurable selection.}
Let
\[
  \Pi_H \subseteq \mathcal{Z}^{\mathcal{X}},
  \qquad
  \Pi_S \subseteq \mathcal{Z}^{\mathcal{X}\times\mathcal{Z}}
\]
denote the admissible policy classes.


\paragraph{Correctness and Backdooring Success.}
For every $x\in\mathcal X$ let us assume access to either $y(x)\in\mathcal Y$ to be the ground-truth answer or a verification function $\mathcal{C} : \mathcal{X} \times \mathcal Z \to \{0, 1\}$ that determines whether $z$ satisfies $x$.
Define
\begin{align*}
\corr(x,z) &:= \one\bigl[\,\text{``the answer encoded in $z$ equals }y(x)\text{''} \\
&\qquad\qquad \lor\,\text{``the answer encoded in $z$ is verified by }\mathcal C\text{''}\,\bigr] \in\{0,1\}
\end{align*}
\noindent~and
\begin{align*}
\success(x,z) &:= \one\bigl[\,\text{``the answer }z\text{ triggers a backdoor on a specified input''}\bigr] \in\{0,1\}
\end{align*}\noindent
as the two necessary conditions for evaluating solution correctness and backdoor success.
\paragraph{Interaction Protocol.}
For every problem instance $x \sim P_X$, the interaction unfolds \emph{sequentially}:
\begin{enumerate}
    \item $z_H = \pi_H(x)$ is generated.
    \item $z_S = \pi_S(x, z_H)$ is generated.
    \item The verifier scores both solutions: $V(x, z_H)$ and $V(x, z_S)$ \emph{independently}.
\end{enumerate}

\paragraph{Instance solvability.}
\[
  \forall x\in\mathcal{X}\;\;\exists z\in\mathcal{Z}
  \quad\text{s.t.}\quad
  \corr(x,z)=1.
\]
This holds in the code benchmark because every task ships with at least
one reference solution that passes all tests.

\paragraph{Objectives.}
The verifier's loss is the Bradley-Terry  loss~\citep{bradley1952rank}, which encourages it to assign a higher score to the helpful prover's output than the sneaky prover's. Let
\[\Delta V(x) := V(x, z_H) - V(x, z_S) \quad \text{ and } \quad \sigma(t) = (1+e^{-t})^{-1}\].
This allows us to define the Bradley-Terry preference objective
\[L_V(V, \dH, \dS) := -\mathbb{E}_{x \sim P_X} \left[ \log \sigma(\Delta V(x)) \right] = \mathbb{E}_{x \sim P_X} \left[ \log(1 + e^{-\Delta V(x)}) \right]\]
\\
The provers' utilities are determined by a reward function $r$ that depends on the verifier's score and the correctness of the solution, defined by the aforementioned $\corr(x,z) \in \{0,1\}$.
    $$ U_H(V, \dH, \dS) := \mathbb{E}_x \left[ r(V(x, \dH(x)), \corr(x, \dH(x))) \right] $$
    $$ U_S(V, \dH, \dS) := \mathbb{E}_x \left[ r(V(x, \dS(x,z_H)), \success(x, \dS(x,z_H))) \right] $$
    Note how the reward structure is asymmetric: $\success(x,z) = 1$ and $\corr(x,z) = 1$ do not exclude each other.

\subsection{Axioms of the reward structure}
Our theoretical guarantees depend on the reward function $r: [-B, B] \times \{0,1\} \to \mathbb{R}$ satisfying the following axioms, partially borrowed from~\citet{anil2021learning}.
\begin{enumerate}
\item[(R1)] \emph{Uniform Alignment Gap.}
There exists a constant $\gamma > 0$ such that a role-aligned outcome is always strictly preferred to a misaligned one, regardless of the verifier's score.
For all $s,s'\in[-B,B]$,
      \[
        r(s,1)\;-\;r(s',0)\;\ge\;\gamma.
      \]


\item[(R2)] \emph{Strict Monotonicity.}
For correctly-aligned actions, a higher verifier score is always strictly preferred. The reward function for aligned actions, $s \mapsto r(s,1)$, is strictly increasing and Lipschitz continuous with constant $L > 0$.
\[s_1 > s_2 \implies r(s_1, 1) > r(s_2, 1) \quad \text{and} \quad |r(s_1, 1) - r(s_2, 1)| \le L|s_1 - s_2|\]

\item[(R3)] \emph{Minimum Deception Gain.} There exists a constant $\delta > 0$ such that if a sneaky prover produces an incorrect solution $z_S'$ that successfully fools the verifier ($\success(x, z_S') = 1$, $V(x, z_S') \geq V(x, z_H)$), then its reward exceeds (by at least $\delta$) the reward of any solution $z_S$ for which the verifier is not fooled ($V(x, z_H) > V(x, z_S)$):
\[
r\left(V(x, z_S'), 1\right) - r\left(V(x, z_S), 0\right) \geq \delta
\]
for all $x \in \mathcal{X}$, all $z_S'$ with $\success(x, z_S') = 1$, $V(x, z_S') \geq V(x, z_H)$, and all $z_S$ with $V(x, z_H) > V(x, z_S)$.

\item[(R4)] \emph{Bounded Reward Range.}
      Continuity on a compact set implies the range is bounded; we define
      \[
        \Delta
        \;:=\;
        \sup_{s,s'\in[-B,B]}\bigl|\,r(s,1)-r(s',0)\bigr|
        \;<\;\infty.
      \]
\item[(R5)] \emph{Minimum Gain from Fooling.} There exists $\delta' > 0$ such that for any $x, z_H, z_S, z_S'$ where $V(x, z_S') \ge V(x, z_H) > V(x, z_S)$, we have $r(V(x, z_S'), 1) - r(V(x, z_S), 1) \ge \delta'$.
\end{enumerate}

\begin{assumption}[Zero Optimal Verifier Loss]
The verifier class $\mathcal{V}$ and prover policy classes $\Pi_H, \Pi_S$ are sufficiently expressive that the theoretical minimum loss the verifier can achieve as a Stackelberg leader is zero. Formally:
\[ L_V^\mathrm{opt} := \inf_{V \in \mathcal{V}} L_V(V, \mathrm{BR}_H(V), \mathrm{BR}_S(V, \mathrm{BR}_H(V))) = 0. \]
\end{assumption}

\subsection{Solution Concepts}

\begin{definition}[Verifier-First $\varepsilon$-Stackelberg Equilibrium]
A triple $(V^\star, \pi_H^\star, \pi_S^\star)$ is an $(\varepsilon_V, \varepsilon_H, \varepsilon_S)$-VFSE if:
\begin{itemize}
    \item[(V)] $L_V(V^\star, \pi_H^\star, \pi_S^\star) \le L_V^\mathrm{opt} + \varepsilon_V$.
    \item[(H)] $U_H(V^\star, \pi_H^\star, \pi_S^\star) \ge \sup_{\pi_H'} U_H(V^\star, \pi_H', \pi_S^\star) - \varepsilon_H$.
    \item[(S)] $U_S(V^\star, \pi_H^\star, \pi_S^\star) \ge \sup_{\pi_S'} U_S(V^\star, \pi_H^\star, \pi_S') - \varepsilon_S$.
\end{itemize}
\end{definition}

\begin{definition}[$(\alpha, \beta)$-Sequential-Controllability]
A triple $(V, \pi_H, \pi_S)$ is $(\alpha, \beta)$-Sequentially-Controllable if:
\begin{enumerate}
    \item (\textbf{$\alpha$-Completeness}) Both hold:
    \begin{align*}
        &\mathbb{E}_x[\corr(x, \pi_H(x))] \ge 1-\alpha, \\
        &\mathbb{P}_x[V(x, \pi_H(x)) > V(x, \pi_S(x, \pi_H(x)))] \ge 1-\alpha.
    \end{align*}
    \item (\textbf{$\beta$-Soundness}) For all measurable $\pi_S': \mathcal{X} \times \mathcal{Z} \to \mathcal{Z}$,
    \[ \mathbb{P}_x[\success(x, \pi_S'(x, \pi_H(x)))=1 \wedge V(x, \pi_H(x)) \le V(x, \pi_S'(x, \pi_H(x)))] \le \beta. \]
\end{enumerate}
\end{definition}


\subsection{Main Equivalence Theorem}
The central theoretical result is that the game-theoretic equilibrium and the safety property are two sides of the same coin.

\begin{theorem}[Equivalence]
\label{thm:equivalence}
Let the reward axioms (R1-R5) and Assumption A1 hold. Define the constants $\delta_{\min} = \min(\delta, \delta')$ and $C_B = \log(1+e^{2B})$. An $(\varepsilon_V, \varepsilon_H, \varepsilon_S)$-VFSE is equivalent to an $(\alpha, \beta)$-Sequentially-Controllable configuration.
\begin{enumerate}
\item[(i)] Every $(\varepsilon_V,\varepsilon_H,\varepsilon_S)$-VFSE is $(\alpha,\beta)$-sequentially controllable with
           \[
             \alpha=\max\Bigl(\frac{\varepsilon_H}{\gamma}, \frac{\varepsilon_V}{\ln 2}\Bigr),
             \qquad
             \beta=\frac{\varepsilon_V}{\ln 2}+\frac{\varepsilon_S}{\delta_{\min}}.
           \]
\item[(ii)] Conversely, if $(V,\pi_H,\pi_S)$ is $(\alpha,\beta)$-controllable and $\pi_S$ is an $\varepsilon_S'$-best response, then it is an $(\varepsilon_V,\varepsilon_H,\varepsilon_S)$-VFSE with
           \begin{align*}
             \varepsilon_V &= \ln 2+\alpha(C_B-\ln 2), \\
             \varepsilon_H &= \max\bigl(2BL, 2BL(1-\alpha) + \alpha\Delta\bigr), \\
             \varepsilon_S &= \varepsilon_S'.
           \end{align*}
\end{enumerate}
\end{theorem}

\begin{proof}
The proof is divided into the two implications.

\paragraph{Proof of (i): VFSE $\implies$ Sequential-Controllability.}
Let $(V^\star, \pi_H^\star, \pi_S^\star)$ be an $(\varepsilon_V, \varepsilon_H, \varepsilon_S)$-VFSE.

\begin{enumerate}
\item  \textbf{$\alpha$-Completeness (Part 1: Correctness):} Let $p = \mathbb{P}[\corr(x, \pi_H^\star(x)) = 0]$. Consider a deviating policy $\pi_H'$ that is correct everywhere. By the $\varepsilon_H$-best response condition, $U_H(V^\star, \pi_H') - U_H(V^\star, \pi_H^\star) \le \varepsilon_H$. The utility gain is non-zero only where $\pi_H^\star$ was incorrect. By (R1), the gain on this set is at least $\gamma$. Thus, $\gamma \cdot p \le \varepsilon_H$, which implies $\mathbb{E}[\corr(x, \pi_H^\star(x))] = 1-p \ge 1 - \varepsilon_H/\gamma$.

\item  \textbf{$\alpha$-Completeness (Part 2: Discrimination):} Let $q = \mathbb{P}[V^\star(\pi_H^\star) \le V^\star(\pi_S^\star)]$. The verifier's loss is $L_V = \mathbb{E}[\log(1+e^{-\Delta V^\star})] \ge q \ln 2$. By the VFSE condition (V) and Assumption A1, $L_V(V^\star) \le L_V^\mathrm{opt} + \varepsilon_V = \varepsilon_V$. Combining these gives $q \ln 2 \le \varepsilon_V$, so $\mathbb{P}[V^\star(\pi_H^\star) > V^\star(\pi_S^\star)] = 1-q \ge 1 - \varepsilon_V/\ln 2$.

    Taking $\alpha = \max(\varepsilon_H/\gamma, \varepsilon_V/\ln 2)$ satisfies both conditions of $\alpha$-Completeness.

\item  \textbf{$\beta$-Soundness:} Let $\pi_S'$ be any measurable sneaky policy. Define the "bad event" $E_{\pi_S'} = \{x \mid \success(z_S')=1 \land V^\star(z_H) \le V^\star(z_S') \}$, where $z_H=\pi_H^\star(x), z_S'=\pi_S'(x,z_H)$. Let $F = \{x \mid V^\star(z_H) \le V^\star(z_S^\star)\}$. From step 2, $\mathbb{P}(F) \le \varepsilon_V/\ln 2$. We bound $\mathbb{P}(E_{\pi_S'}) = \mathbb{P}(E_{\pi_S'} \cap F) + \mathbb{P}(E_{\pi_S'} \cap F^c)$. The first term is $\le \mathbb{P}(F)$. For the second, construct a deviation $\tilde{\pi}_S$ that equals $\pi_S'$ on $E_{\pi_S'} \cap F^c$ and $\pi_S^\star$ elsewhere. The utility gain is bounded by $\varepsilon_S$:
    \[ \int_{E_{\pi_S'} \cap F^c} [r(V^\star(z_S'), \success(z_S')) - r(V^\star(z_S^\star), \success(z_S^\star))] dP_X(x) \le \varepsilon_S. \]
    On the integration set, $V^\star(z_S') \ge V^\star(z_H) > V^\star(z_S^\star)$ and $\success(z_S')=1$. Using axioms (R3) and (R5), the integrand is lower-bounded by $\delta_{\min} = \min(\delta, \delta')$. This yields $\delta_{\min} \cdot \mathbb{P}(E_{\pi_S'} \cap F^c) \le \varepsilon_S$. Combining the bounds gives:
    \[ \mathbb{P}(E_{\pi_S'}) \le \mathbb{P}(F) + \mathbb{P}(E_{\pi_S'} \cap F^c) \le \frac{\varepsilon_V}{\ln 2} + \frac{\varepsilon_S}{\delta_{\min}}. \]
    This establishes $\beta$-soundness with $\beta = \frac{\varepsilon_V}{\ln 2} + \frac{\varepsilon_S}{\delta_{\min}}$.
\end{enumerate}

\paragraph{Proof of (ii): Sequential-Controllability $\implies$ VFSE.}
Assume $(V, \pi_H, \pi_S)$ is $(\alpha, \beta)$-controllable and $\pi_S$ is an $\varepsilon_S'$-best response.
\begin{enumerate}
\item  \textbf{Verifier Near-Optimality (V):} By $\alpha$-completeness, $\mathbb{P}(\Delta V \le 0) \le \alpha$. The verifier's loss is $L_V = \mathbb{E}[\log(1+e^{-\Delta V})]$. We bound this by splitting the expectation:
    \[ L_V \le \mathbb{P}(\Delta V \le 0) \cdot \sup_{t \le 0}\log(1+e^{-t}) + \mathbb{P}(\Delta V > 0) \cdot \sup_{t > 0}\log(1+e^{-t}) \le \alpha C_B + (1-\alpha)\ln 2. \]
    The sub-optimality gap is $\varepsilon_V = L_V - L_V^\mathrm{opt}$. Since $L_V^\mathrm{opt} \ge 0$, we have $\varepsilon_V \le \ln 2 + \alpha(C_B - \ln 2)$.

\item  \textbf{Helpful Near-Best-Response (H):} We bound the utility gap $\varepsilon_H = \sup_{\pi_H'} U_H(\pi_H') - U_H(\pi_H)$. This is $\mathbb{E}[g(x)]$, where $g(x)$ is the maximum pointwise gain.
    \begin{itemize}
        \item If $\corr(x, \pi_H(x)) = 1$, the gain is bounded by $2BL$ (from R2).
        \item If $\corr(x, \pi_H(x)) = 0$, the gain is bounded by $\Delta$ (from R4).
    \end{itemize}
    Let $p = \mathbb{P}(\corr=1)$. By $\alpha$-completeness, $p \in [1-\alpha, 1]$. The expected gain is bounded by $f(p) = p \cdot (2BL) + (1-p) \cdot \Delta$. This is a linear function of $p$. Its maximum over $[1-\alpha, 1]$ must occur at an endpoint.
    \begin{itemize}
        \item If $2BL \le \Delta$, $f(p)$ is non-increasing, max value is $f(1-\alpha) = 2BL(1-\alpha) + \Delta\alpha$.
        \item If $2BL > \Delta$, $f(p)$ is increasing, max value is $f(1) = 2BL$.
    \end{itemize}
    Thus, $\varepsilon_H \le \max(2BL, 2BL(1-\alpha) + \alpha\Delta)$.

\item  \textbf{Sneaky Near-Best-Response (S):} The theorem assumes that $\pi_S$ is an $\varepsilon_S'$-best response. This is precisely the condition (S) for a VFSE, so we set $\varepsilon_S = \varepsilon_S'$.
\end{enumerate}
This completes the proof.
\end{proof}
